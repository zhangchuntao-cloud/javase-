	    public static int count = 3;
	    public static BigDecimal price = BigDecimal.valueOf(100.0);//定义的时候就用BigDecimal来确保两位小数
	    public static BigDecimal money = BigDecimal.valueOf(0.0);
	    //调用BigDecimal来确保两位小数
	    static final BigDecimal MIN = BigDecimal.valueOf(0.01);
	    public void run(){
	        synchronized (MyThread4.class) {
	            if(count==0){
	                System.out.println(getName()+"没有抢到红包!!");
	            }else{
	                if(count==1){
	                    money = price;
	                }else{
	                    //生成随机红包金额
	                    BigDecimal max = price.subtract(BigDecimal.valueOf(count-1).multiply(MIN));
	
	                    Random r = new Random();
	                    money = BigDecimal.valueOf(r.nextDouble(max.doubleValue()));
	
	
	                }
	                money = money.setScale(2, BigDecimal.ROUND_HALF_UP);
	                price =  price.subtract(money);
	                count--;
	                System.out.println(getName()+"抢到了"+money+"元");
	            }
	
	        }
	    }
	
	
	
	}

	    public static void UDP1() throws IOException {
	        //udp协议(发送数据)
	        //1.创建DatagramSocket对象
	        //细节:绑定端口, 以后我们就是通过这个端口往外发送
	        //有参就是指定端口号进行绑定, 无参就是所有可用的端口中随机一个进行使用
	        DatagramSocket ds = new DatagramSocket();//无参构造
	        /*这是 无参构造，系统会自动从「1024~65535」的可用端口中随机分配一个端口，作为当前 UDP 发送端的绑定端口；
	代码中后面的 int port = 10000 是 接收端的端口号（用于指定数据要发给哪个端口），和 DatagramSocket 自身的绑定端口是两回事！*/
	
	        //2.打包数据
	        String str = "hello";
	        byte[] bytes = str.getBytes();
	        InetAddress address = InetAddress.getByName("127.0.0.1");
	        int port = 10000;//接收端的端口号
	
	        DatagramPacket packet = new DatagramPacket(bytes, bytes.length, address, port);
	        //3.发送数据
	        ds.send(packet);
	        //4.关闭资源
	        ds.close();
	    }

	interface Star{

	class ProxyUtil{
	    //创建一个代理
	    public static Star createProxy(BigStar bigStar){
	        //给一个明星对象, 创建一个代理
	        Star star = (Star) Proxy.newProxyInstance(
	                ProxyUtil.class.getClassLoader(),//1.用于指定用哪个类的加载器, 去加载生成的代理类
	                new Class[]{Star.class}, //2.能代理的方法都在接口中
	                //3.用于指定生成的代理对象要干什么事情
	                new InvocationHandler() {
	                    @Override
	                    public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
	                        //参数1:代理的对象
	                        //参数2:要运行的方法:sing
	                        //参数3:调用sing方法时要传递的参数
	                        if("sing".equals(method.getName())){
	                            System.out.println("准备话筒, 收钱");
	                        }else if("dance".equals(method.getName())){
	                            System.out.println("准备场地, 收钱");
	                        }
	
	
	                        return method.invoke(bigStar, args);
	                    }
	                }
	                );
	        return star;
	    }

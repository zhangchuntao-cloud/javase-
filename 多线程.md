class MyThread1 extends Thread{
    public MyThread1() {
    }

    public MyThread1(String name) {
        super(name);
    }

    public static int ticket = 1000;
    public void run(){
        while(ticket>1){
            synchronized (MyThread1.class) {
                try {
                    Thread.sleep(3);
                } catch (InterruptedException e) {
                    throw new RuntimeException(e);
                }
                ticket--;
                System.out.println(getName()+"卖出了电影票, 还剩"+ticket+"张");
            }
        }
    }

}

class MyThread4 extends Thread{
    public MyThread4() {
    }

    public MyThread4(String name) {
        super(name);
    }
    public static int count = 3;
    public static BigDecimal price = BigDecimal.valueOf(100.0);//定义的时候就用BigDecimal来确保两位小数
    public static BigDecimal money = BigDecimal.valueOf(0.0);
    //调用BigDecimal来确保两位小数
    static final BigDecimal MIN = BigDecimal.valueOf(0.01);
    public void run(){
        synchronized (MyThread4.class) {
            if(count==0){
                System.out.println(getName()+"没有抢到红包!!");
            }else{
                if(count==1){
                    money = price;
                }else{
                    //生成随机红包金额
                    BigDecimal max = price.subtract(BigDecimal.valueOf(count-1).multiply(MIN));

                    Random r = new Random();
                    money = BigDecimal.valueOf(r.nextDouble(max.doubleValue()));


                }
                money = money.setScale(2, BigDecimal.ROUND_HALF_UP);
                price =  price.subtract(money);
                count--;
                System.out.println(getName()+"抢到了"+money+"元");
            }

        }
    }



}
class MyThread5 extends Thread{
    private static ArrayList<Integer> list;

    public MyThread5() {
    }

    public MyThread5(String name, ArrayList<Integer> list) {
        super(name);
        this.list = list;
    }
    public void run(){
        while(list.size()>0) {
            synchronized (MyThread5.class) {
                Collections.shuffle(list);
                System.out.println(getName() + " 又产生了一个" + list.remove(0) + "大奖");
            }
            //睡10毫秒,要放在锁的外面,因为睡的目的是让其他线程有机会进入锁, 放在锁里面会导致其他线程一直等待
            try {
                Thread.sleep(10);
            } catch (InterruptedException e) {
                throw new RuntimeException(e);
            }
        }

    }
}

class MyCallable implements Callable<Integer> {
    public MyCallable() {
    }

    ArrayList<Integer> list;
    public MyCallable(ArrayList<Integer> list) {
        this.list = list;
    }

    public Integer call() throws Exception {
        ArrayList<Integer> box = new ArrayList<>();
        while (true){
            synchronized (MyCallable.class) {
                if(list.size()==0){
                    System.out.println(Thread.currentThread().getName()+" "+box);
                    break;//跳出while循环
                }else{
                    Collections.shuffle(list);
                    box.add(list.remove(0));//从list中移除一个元素,并添加到box中
                }
            }
            //睡10毫秒,要放在锁的外面,因为睡的目的是让其他线程有机会进入锁, 放在锁里面会导致其他线程一直等待
            try {
                Thread.sleep(10);
            } catch (InterruptedException e) {
                throw new RuntimeException(e);
            }
        }
        return box.size()==0?0:box.stream().max(Integer::compareTo).get();
    }
}
